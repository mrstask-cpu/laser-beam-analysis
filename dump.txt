def inverse_abel_basex(
    img_array,
    origin_x,
    regularization=50,
    symmetry_axis=0,
    debug_plot=True
):
    """
    Inverse Abel transform методом BASEX.

    Parameters
    ----------
    img_array : np.ndarray
        Предварительно обработанная 2D projection image.

    origin_x : float
        Найденная координата вертикальной оси симметрии.

    regularization : float
        Tikhonov regularization parameter для BASEX.

    symmetry_axis : int
        0 означает симметрию относительно вертикальной оси,
        то есть левую/правую часть изображения.

    debug_plot : bool
        Показывать результат.

    Returns
    -------
    reconstructed : np.ndarray
        Восстановленное 2D распределение I(r,z).

    transform : abel.Transform
        Объект PyAbel с дополнительной информацией.
    """

    import abel

    img_array = np.asarray(
        img_array,
        dtype=np.float64
    )

    if img_array.ndim != 2:
        raise ValueError(
            "Для Abel inversion нужен 2D-массив."
        )

    if not np.all(np.isfinite(img_array)):
        raise ValueError(
            "Входное изображение содержит NaN или inf."
        )

    if regularization < 0:
        raise ValueError(
            "regularization не может быть отрицательным."
        )

    # PyAbel принимает origin в формате (y, x).
    #
    # Мы точно знаем только X-координату,
    # поэтому Y оставляем None.
    origin = (
        None,
        float(origin_x)
    )

    print("=== Abel inversion: BASEX ===")
    print(f"Метод:               BASEX")
    print(f"Origin X:            {origin_x:.5f} px")
    print(f"Regularization:      {regularization}")
    print()

    transform = abel.Transform(
        img_array,
        direction="inverse",
        method="basex",

        # Точный найденный центр
        origin=origin,

        # Для Abel физически используем левую/правую симметрию.
        symmetry_axis=symmetry_axis,

        # Усредняем обе половины.
        symmetrize_method="average",

        # Требуется для BASEX.
        center_options={
            "odd_size": True,
            "axes": 1,
            "crop": "valid_region"
        },

        # Параметры BASEX.
        transform_options={
            "reg": regularization
        },

        recast_as_float64=True,
        verbose=True
    )

    reconstructed = transform.transform

    # ------------------------------------------------------------
    # Небольшая защита от численного отрицательного хвоста
    # ------------------------------------------------------------

    negative_fraction = np.mean(
        reconstructed < 0
    )

    min_value = np.min(
        reconstructed
    )

    max_value = np.max(
        reconstructed
    )

    print("=== Результат BASEX ===")
    print(f"Размер результата:    {reconstructed.shape}")
    print(f"Минимум:              {min_value:.6g}")
    print(f"Максимум:             {max_value:.6g}")
    print(
        f"Доля отрицательных "
        f"пикселей:             {negative_fraction:.6%}"
    )
    print()

    if debug_plot:

        fig, axes = plt.subplots(
            1,
            2,
            figsize=(14, 5)
        )

        axes[0].imshow(
            img_array,
            cmap="viridis",
            aspect="auto"
        )

        axes[0].set_title(
            "Projection"
        )

        axes[0].set_xlabel("X, px")
        axes[0].set_ylabel("Z, px")

        im = axes[1].imshow(
            reconstructed,
            cmap="viridis",
            aspect="auto"
        )

        axes[1].set_title(
            "Inverse Abel / BASEX"
        )

        axes[1].set_xlabel("r, px")
        axes[1].set_ylabel("Z, px")

        plt.colorbar(
            im,
            ax=axes[1]
        )

        plt.tight_layout()
        plt.show()

    return reconstructed, transform

def basex_regularization_sweep(
    img_array,
    origin_x,
    regularization_values=None,
    debug_plot=True
):
    """
    Исследует влияние регуляризации BASEX.

    Для каждого значения рассчитывается:
        - reconstructed image
        - максимум
        - минимум
        - доля отрицательных значений
        - минимальная относительная амплитуда
        - оценка гладкости

    Это исследовательский инструмент, а не финальный auto-selector.
    """

    if regularization_values is None:
        regularization_values = [
            0,
            1,
            5,
            10,
            25,
            50,
            100,
            200
        ]

    import abel

    results = {}

    for reg in regularization_values:

        print(
            f"\n=== BASEX regularization = {reg} ==="
        )

        transform = abel.Transform(
            img_array,
            direction="inverse",
            method="basex",

            origin=(
                None,
                float(origin_x)
            ),

            symmetry_axis=0,
            symmetrize_method="average",

            center_options={
                "odd_size": True,
                "axes": 1,
                "crop": "valid_region"
            },

            transform_options={
                "reg": float(reg)
            },

            recast_as_float64=True,
            verbose=False
        )

        recon = transform.transform

        maximum = np.max(recon)
        minimum = np.min(recon)

        negative_fraction = np.mean(
            recon < 0
        )

        relative_negative_amplitude = (
            abs(minimum) / maximum
            if maximum > 0
            else np.nan
        )

        # --------------------------------------------------------
        # Оценка шероховатости.
        #
        # Используем вторую производную только по X.
        # Большие значения означают сильные локальные осцилляции.
        # --------------------------------------------------------

        second_derivative = np.diff(
            recon,
            n=2,
            axis=1
        )

        roughness = np.sqrt(
            np.mean(
                second_derivative ** 2
            )
        )

        results[reg] = {
            "transform": transform,
            "image": recon,
            "max": maximum,
            "min": minimum,
            "negative_fraction": negative_fraction,
            "relative_negative_amplitude":
                relative_negative_amplitude,
            "roughness": roughness
        }

        print(
            f"min = {minimum:.6g}"
        )

        print(
            f"max = {maximum:.6g}"
        )

        print(
            f"negative fraction = "
            f"{negative_fraction:.3%}"
        )

        print(
            f"relative negative amplitude = "
            f"{relative_negative_amplitude:.4%}"
        )

        print(
            f"roughness = "
            f"{roughness:.6g}"
        )

    if debug_plot:

        regs = np.array(
            list(results.keys()),
            dtype=float
        )

        neg_amp = np.array([
            results[r]["relative_negative_amplitude"]
            for r in results
        ])

        roughness = np.array([
            results[r]["roughness"]
            for r in results
        ])

        fig, axes = plt.subplots(
            1,
            2,
            figsize=(14, 5)
        )

        axes[0].plot(
            regs,
            neg_amp,
            marker="o"
        )

        axes[0].set_xlabel(
            "BASEX regularization"
        )

        axes[0].set_ylabel(
            "Relative negative amplitude"
        )

        axes[0].set_title(
            "Негативный численный хвост"
        )

        axes[0].grid(
            True,
            alpha=0.3
        )

        axes[1].plot(
            regs,
            roughness,
            marker="o"
        )

        axes[1].set_xlabel(
            "BASEX regularization"
        )

        axes[1].set_ylabel(
            "Roughness"
        )

        axes[1].set_title(
            "Гладкость реконструкции"
        )

        axes[1].grid(
            True,
            alpha=0.3
        )

        plt.tight_layout()
        plt.show()

    return results

def inverse_abel_hansenlaw(
    img_array,
    origin_x,
    symmetry_axis=0,
    debug_plot=True
):
    """
    Inverse Abel transform методом Hansen-Law.
    Используется как независимая численная проверка BASEX.
    """

    import abel

    img_array = np.asarray(
        img_array,
        dtype=np.float64
    )

    if img_array.ndim != 2:
        raise ValueError(
            "Для Abel inversion нужен 2D-массив."
        )

    if not np.all(np.isfinite(img_array)):
        raise ValueError(
            "Изображение содержит NaN или inf."
        )

    transform = abel.Transform(
        img_array,
        direction="inverse",
        method="hansenlaw",

        origin=(
            None,
            float(origin_x)
        ),

        symmetry_axis=symmetry_axis,
        symmetrize_method="average",

        center_options={
            "odd_size": True,
            "axes": 1,
            "crop": "valid_region"
        },

        recast_as_float64=True,
        verbose=True
    )

    reconstructed = transform.transform

    print("=== Результат Hansen-Law ===")
    print(
        f"Размер результата: "
        f"{reconstructed.shape}"
    )

    print(
        f"Диапазон: "
        f"{reconstructed.min():.6g} ... "
        f"{reconstructed.max():.6g}"
    )

    if debug_plot:

        plt.figure(
            figsize=(8, 5)
        )

        plt.imshow(
            reconstructed,
            cmap="viridis",
            aspect="auto"
        )

        plt.title(
            "Inverse Abel / Hansen-Law"
        )

        plt.xlabel("r, px")
        plt.ylabel("Z, px")

        plt.colorbar(
            label="Intensity"
        )

        plt.tight_layout()
        plt.show()

    return reconstructed, transform
