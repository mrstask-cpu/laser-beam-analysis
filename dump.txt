def preprocess_background_and_noise(
    img_array,
    bg_corner_size=40,
    sigma=1.5,
    debug_plot=True
):
    """
    Научная предобработка изображения:

        1. оценка постоянного фона;
        2. вычитание фона;
        3. Gaussian smoothing;
        4. оценка остаточного шума.

    ВАЖНО:
    отрицательные значения после вычитания фона НЕ обрезаются.
    Они являются допустимым представлением шумового отклонения.

    Отрицательные значения можно обрезать только при визуализации
    или при формировании пользовательского display-массива.

    Parameters
    ----------
    img_array : np.ndarray
        Геометрически выровненное изображение.

    bg_corner_size : int
        Размер области в каждом углу для оценки фона.

    sigma : float
        Sigma Gaussian filter в пикселях.

    debug_plot : bool
        Показывать диагностические графики.

    Returns
    -------
    filtered_image : np.ndarray

    info : dict
    """

    img_array = np.asarray(
        img_array,
        dtype=np.float64
    )

    if img_array.ndim != 2:
        raise ValueError(
            "img_array должен быть двумерным."
        )

    if not np.all(np.isfinite(img_array)):
        raise ValueError(
            "img_array содержит NaN или inf."
        )

    if bg_corner_size <= 0:
        raise ValueError(
            "bg_corner_size должен быть > 0."
        )

    if sigma < 0:
        raise ValueError(
            "sigma не может быть отрицательной."
        )

    h, w = img_array.shape

    if 2 * bg_corner_size > min(h, w):
        raise ValueError(
            "bg_corner_size слишком велик."
        )

    # ============================================================
    # 1. Выбор угловых областей
    # ============================================================

    corners = np.concatenate([
        img_array[
            :bg_corner_size,
            :bg_corner_size
        ].ravel(),

        img_array[
            :bg_corner_size,
            -bg_corner_size:
        ].ravel(),

        img_array[
            -bg_corner_size:,
            :bg_corner_size
        ].ravel(),

        img_array[
            -bg_corner_size:,
            -bg_corner_size:
        ].ravel()
    ])

    # ============================================================
    # 2. Оценка фона и шума
    # ============================================================

    bg_level = np.median(corners)

    mad_before = np.median(
        np.abs(
            corners - bg_level
        )
    )

    noise_std_before = (
        1.4826 * mad_before
    )

    conventional_std_before = np.std(
        corners
    )

    # ============================================================
    # 3. Вычитание фона
    # ============================================================

    # НЕ используем clip!
    img_subtracted = (
        img_array - bg_level
    )

    # ============================================================
    # 4. Gaussian smoothing
    # ============================================================

    if sigma > 0:

        img_filtered = gaussian_filter(
            img_subtracted,
            sigma=sigma
        )

    else:

        img_filtered = img_subtracted.copy()

    # ============================================================
    # 5. Оценка остаточного шума
    # ============================================================

    corners_filtered = np.concatenate([
        img_filtered[
            :bg_corner_size,
            :bg_corner_size
        ].ravel(),

        img_filtered[
            :bg_corner_size,
            -bg_corner_size:
        ].ravel(),

        img_filtered[
            -bg_corner_size:,
            :bg_corner_size
        ].ravel(),

        img_filtered[
            -bg_corner_size:,
            -bg_corner_size:
        ].ravel()
    ])

    filtered_bg_median = np.median(
        corners_filtered
    )

    mad_after = np.median(
        np.abs(
            corners_filtered -
            filtered_bg_median
        )
    )

    noise_std_after = (
        1.4826 * mad_after
    )

    if noise_std_after > 0:

        noise_reduction_factor = (
            noise_std_before /
            noise_std_after
        )

    else:

        noise_reduction_factor = np.inf

    # ============================================================
    # 6. Signal metrics
    # ============================================================

    max_before = np.max(
        img_array
    )

    max_after = np.max(
        img_filtered
    )

    # ============================================================
    # 7. Output
    # ============================================================

    info = {
        "background": bg_level,

        "noise_std_before":
            noise_std_before,

        "noise_std_after":
            noise_std_after,

        "conventional_std_before":
            conventional_std_before,

        "noise_reduction_factor":
            noise_reduction_factor,

        "max_before":
            max_before,

        "max_after":
            max_after,

        "sigma":
            sigma
    }

    print("=== Анализ фона и шума ===")
    print(
        f"Медианный фон:                "
        f"{bg_level:.3f}"
    )

    print(
        f"СКО шума до:                  "
        f"{noise_std_before:.3f}"
    )

    print(
        f"Обычное СКО углов:            "
        f"{conventional_std_before:.3f}"
    )

    print(
        f"СКО шума после:               "
        f"{noise_std_after:.3f}"
    )

    print(
        f"Коэффициент подавления шума:  "
        f"{noise_reduction_factor:.2f}"
    )

    print(
        f"Максимум до обработки:        "
        f"{max_before:.3f}"
    )

    print(
        f"Максимум после обработки:     "
        f"{max_after:.3f}"
    )

    print(
        f"Gaussian sigma:                "
        f"{sigma:.3f}"
    )

    print()

    # ============================================================
    # 8. Диагностика
    # ============================================================

    if debug_plot:

        # Для отображения можно отдельно ограничить
        # отрицательные значения нулём.
        display_image = np.clip(
            img_filtered,
            0,
            None
        )

        fig, axes = plt.subplots(
            1,
            2,
            figsize=(13, 5)
        )

        im1 = axes[0].imshow(
            img_subtracted,
            cmap="viridis",
            aspect="auto"
        )

        axes[0].set_title(
            "После вычитания фона"
        )

        axes[0].set_xlabel("X, px")
        axes[0].set_ylabel("Y, px")

        plt.colorbar(
            im1,
            ax=axes[0]
        )

        im2 = axes[1].imshow(
            display_image,
            cmap="viridis",
            aspect="auto"
        )

        axes[1].set_title(
            rf"После Gaussian $\sigma={sigma}$"
        )

        axes[1].set_xlabel("X, px")
        axes[1].set_ylabel("Y, px")

        plt.colorbar(
            im2,
            ax=axes[1]
        )

        plt.tight_layout()
        plt.show()

        # Центральное сечение

        mid_y = h // 2

        plt.figure(
            figsize=(9, 4)
        )

        plt.plot(
            img_array[mid_y],
            label="Исходное"
        )

        plt.plot(
            img_filtered[mid_y],
            label="Обработанное"
        )

        plt.axhline(
            0,
            linestyle="--",
            linewidth=1
        )

        plt.xlabel("X, px")
        plt.ylabel("Intensity")

        plt.title(
            "Центральное поперечное сечение"
        )

        plt.grid(
            True,
            alpha=0.3
        )

        plt.legend()
        plt.tight_layout()
        plt.show()

    return img_filtered, info
