def compare_basex_hansenlaw(
    basex_results,
    hansenlaw_results
):
    """
    Сравнивает BASEX и Hansen–Law
    по успешным Gaussian fits.

    Если отдельное сечение не удалось обработать,
    оно исключается только из сравнения этого сечения.
    """

    basex_map = {
        item["z_index"]: item
        for item in basex_results
        if item.get("success", False)
    }

    hl_map = {
        item["z_index"]: item
        for item in hansenlaw_results
        if item.get("success", False)
    }

    common_z = sorted(
        set(basex_map) &
        set(hl_map)
    )

    if len(common_z) < 2:

        raise RuntimeError(
            "Недостаточно общих успешных "
            "сечений для сравнения."
        )

    d_basex = np.array([
        basex_map[z]["diameter_1e2"]
        for z in common_z
    ])

    d_hl = np.array([
        hl_map[z]["diameter_1e2"]
        for z in common_z
    ])

    diff_percent = (
        100.0
        * np.abs(d_basex - d_hl)
        / np.maximum(0.5 * (np.abs(d_basex) + np.abs(d_hl)), np.finfo(float).eps)
    )

    mean_difference = np.mean(
        diff_percent
    )

    max_difference = np.max(
        diff_percent
    )

    print(
        "=== BASEX vs Hansen–Law ==="
    )

    print(
        f"Сравнено сечений: "
        f"{len(common_z)}"
    )

    print(
        f"Среднее абсолютное различие: "
        f"{mean_difference:.3f}%"
    )

    print(
        f"Максимальное абсолютное различие: "
        f"{max_difference:.3f}%"
    )

    # ------------------------------------------------------------
    # График диаметров
    # ------------------------------------------------------------

    plt.figure(
        figsize=(9, 5)
    )

    plt.plot(
        common_z,
        d_basex,
        marker="o",
        label="BASEX"
    )

    plt.plot(
        common_z,
        d_hl,
        marker="o",
        label="Hansen–Law"
    )

    plt.xlabel(
        "Продольная координата, px"
    )

    plt.ylabel(
        "D 1/e², px"
    )

    plt.title(
        "BASEX vs Hansen–Law"
    )

    plt.grid(
        True,
        alpha=0.3
    )

    plt.legend()
    plt.tight_layout()
    plt.show()

    # ------------------------------------------------------------
    # График различия
    # ------------------------------------------------------------

    plt.figure(
        figsize=(9, 5)
    )

    plt.plot(
        common_z,
        diff_percent,
        marker="o"
    )

    plt.xlabel(
        "Продольная координата, px"
    )

    plt.ylabel(
        "Различие, %"
    )

    plt.title(
        "Различие D 1/e²"
    )

    plt.grid(
        True,
        alpha=0.3
    )

    plt.tight_layout()
    plt.show()

    return {
        "z": np.asarray(
            common_z
        ),

        "diameter_basex":
            d_basex,

        "diameter_hansenlaw":
            d_hl,

        "relative_difference_percent":
            diff_percent,

        "mean_difference_percent":
            mean_difference,

        "max_difference_percent":
            max_difference
    }
