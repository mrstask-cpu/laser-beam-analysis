def make_z_sampling(
    n_z,
    n_points=61,
    margin=10
):
    """
    Создаёт равномерную выборку продольных координат.

    Parameters
    ----------
    n_z : int
        Полное количество доступных продольных строк.

    n_points : int
        Количество выбираемых z-сечений.

    margin : int
        Сколько строк исключить с каждого края.

    Returns
    -------
    np.ndarray
        Отсортированные уникальные z-индексы.
    """

    n_z = int(n_z)
    n_points = int(n_points)
    margin = int(margin)

    if n_z < 3:
        raise ValueError(
            "n_z должен быть >= 3."
        )

    if n_points < 2:
        raise ValueError(
            "n_points должен быть >= 2."
        )

    if margin < 0:
        raise ValueError(
            "margin не может быть отрицательным."
        )

    first = margin
    last = n_z - 1 - margin

    if first > last:
        raise ValueError(
            f"Слишком большой margin={margin}. "
            f"Для изображения с n_z={n_z} "
            f"допустимый диапазон margin: "
            f"0...{(n_z - 1) // 2}."
        )

    available_positions = (
        last - first + 1
    )

    if n_points > available_positions:
        raise ValueError(
            f"Нельзя выбрать {n_points} уникальных "
            f"z-позиций из {available_positions} доступных. "
            f"Уменьшите n_points или margin."
        )

    positions = np.rint(
        np.linspace(
            first,
            last,
            n_points
        )
    ).astype(int)

    positions = np.unique(
        positions
    )

    if len(positions) != n_points:
        raise RuntimeError(
            "При округлении координат появились дубли. "
            "Уменьшите n_points."
        )

    return positions


def get_z_positions(
    n_z,
    mode="dense",
    n_points=61,
    margin=10
):
    """
    Унифицированный выбор продольных координат.

    mode:
        "diagnostic" -> 15 точек
        "dense"      -> n_points
        "full"       -> все допустимые точки
    """

    mode = str(mode).lower()

    if mode == "diagnostic":

        return make_z_sampling(
            n_z=n_z,
            n_points=min(15, n_z - 2 * margin),
            margin=margin
        )

    elif mode == "dense":

        return make_z_sampling(
            n_z=n_z,
            n_points=n_points,
            margin=margin
        )

    elif mode == "full":

        first = int(margin)
        last = int(n_z - 1 - margin)

        if first > last:
            raise ValueError(
                f"margin={margin} слишком велик "
                f"для n_z={n_z}."
            )

        return np.arange(
            first,
            last + 1,
            dtype=int
        )

    else:

        raise ValueError(
            "Неизвестный режим z sampling. "
            "Используйте: "
            "'diagnostic', 'dense' или 'full'."
        )

def analyze_all_z_profiles(
    abel_image,
    z_positions,
    fit_fraction=0.05,
    robust=True,
    r0_max=3.0
):
    """
    Анализирует Gaussian-профиль для каждой z-позиции.

    Неудачные fits не останавливают весь анализ.
    Для них численные параметры записываются как NaN.
    """

    z_positions = np.asarray(
        z_positions,
        dtype=int
    )

    results = []

    metric_names = [
        "amplitude",
        "peak_intensity",
        "r0",
        "sigma",
        "baseline",

        "radius_1e",
        "radius_1e2",

        "diameter_1e",
        "diameter_1e2",

        "beam_radius_w",
        "beam_diameter_2w",

        "fwhm",

        "r_squared",
        "r_squared_signal",
        "r_squared_full",

        "rmse",
        "rmse_signal",
        "rmse_full",

        "nrmse_signal",
        "nrmse_full",

        "max_abs_residual_signal",
        "max_abs_residual_full"
    ]

    for z_index in z_positions:

        result = {
            "z_index": int(z_index),
            "success": False,
            "error": None
        }

        # Заполняем числовые поля NaN,
        # чтобы последующие графики были устойчивыми.
        for name in metric_names:
            result[name] = np.nan

        try:

            r, profile = (
                extract_radial_profile(
                    abel_image,
                    z_index=z_index
                )
            )

            fit_result = (
                fit_gaussian_radial_profile(
                    r,
                    profile,
                    fit_fraction=fit_fraction,
                    robust=robust,
                    r0_max=r0_max
                )
            )

            result.update(
                fit_result
            )

            result["success"] = True

        except Exception as exc:

            result["error"] = str(exc)

            print(
                f"Предупреждение: "
                f"z={z_index}: {exc}"
            )

        results.append(
            result
        )

    valid_count = sum(
        item["success"]
        for item in results
    )

    failed_count = (
        len(results) -
        valid_count
    )

    print()
    print("=== BATCH GAUSSIAN ANALYSIS ===")
    print(
        f"Всего сечений:       {len(results)}"
    )
    print(
        f"Успешных fits:       {valid_count}"
    )
    print(
        f"Неудачных fits:      {failed_count}"
    )

    if valid_count == 0:

        raise RuntimeError(
            "Ни одного успешного Gaussian fit."
        )

    return results
